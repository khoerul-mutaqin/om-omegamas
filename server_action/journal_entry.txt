# To return an action, assign: action = {...}

self = records
# self = records (Terdiri dari Account Move 1 dan Account Move 2)

# 1. Kumpulkan semua baris dari semua dokumen yang dipilih
all_lines = self.mapped('line_ids')

# 2. Filter baris:
# - Harus akun yang 'Allow Reconciliation'
# - Belum punya Matching Number
# - Move harus sudah 'Posted'
reconcilable_lines = all_lines.filtered(
    lambda l: l.account_id.reconcile and not l.full_reconcile_id and l.move_id.state == 'posted'
)

# 3. Ambil daftar akun unik (misal: Stock Interim In/Out)
accounts = reconcilable_lines.mapped('account_id')

for acc in accounts:
    # Filter baris yang hanya menggunakan akun ini
    lines_for_this_account = reconcilable_lines.filtered(lambda l: l.account_id == acc)
    
    # 4. Validasi Saldo Gabungan
    # Jika Move 1 (Debit 100) dan Move 2 (Kredit 100), maka total_balance = 0
    total_balance = sum(lines_for_this_account.mapped('balance'))
    
    # Cek apakah ada lebih dari 1 baris untuk dicocokkan
    if len(lines_for_this_account) > 1:
        if abs(total_balance) < 0.0001:
            # Jika Balance NOL, Odoo akan memberikan 1 Matching Number (misal: A1)
            # yang akan muncul di Move 1 dan Move 2 sekaligus
            lines_for_this_account.reconcile()
        else:
            raise UserError(lines_for_this_account)
            # Jika TIDAK BALANCE, Odoo hanya akan membuat Partial Reconcile 
            # (Tidak ada nomor A1 yang muncul di kolom utama)
            lines_for_this_account.reconcile()
